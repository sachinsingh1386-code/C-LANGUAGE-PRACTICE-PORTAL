21. Calculate age difference between current date and birth date
Code:
#include <stdio.h>
void main(){
    int cd, cm, cy, bd, bm, by;
    printf("Enter current date (dd mm yyyy): ");
    scanf("%d %d %d", &cd, &cm, &cy);
    printf("Enter birth date (dd mm yyyy): ");
    scanf("%d %d %d", &bd, &bm, &by);

    if (cd < bd) { 
        cd += 30;
        cm--; 
    }
    if (cm < bm) { 
        cm += 12;
        cy--; 
    }

    printf("difference = %d years %d months %d days\n", cy - by, cm - bm, cd - bd);
}
Output:
Enter current date (dd mm yyyy): 06 11 2025
Enter birth date (dd mm yyyy): 21 04 2007
difference = 18 years 6 months 15 days
